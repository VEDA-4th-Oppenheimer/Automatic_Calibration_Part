cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(manual_marker_calibration LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  enable_testing()
endif()

find_package(OpenCV REQUIRED COMPONENTS core imgcodecs imgproc calib3d objdetect aruco)
find_package(nlohmann_json REQUIRED)

add_library(manual_marker_core src/marker_calibration.cpp)
target_include_directories(manual_marker_core PUBLIC include)
target_link_libraries(manual_marker_core PUBLIC opencv_core opencv_imgcodecs opencv_imgproc opencv_calib3d opencv_objdetect opencv_aruco nlohmann_json::nlohmann_json)
target_compile_options(manual_marker_core PRIVATE -Wall -Wextra -Wpedantic)

add_executable(generate_charuco_board apps/generate_charuco_board.cpp)
target_link_libraries(generate_charuco_board PRIVATE manual_marker_core)
add_executable(calibrate_camera_markers apps/calibrate_camera_markers.cpp)
target_link_libraries(calibrate_camera_markers PRIVATE manual_marker_core)
add_executable(estimate_marker_pose apps/estimate_marker_pose.cpp)
target_link_libraries(estimate_marker_pose PRIVATE manual_marker_core)
add_executable(compare_marker_to_automatic apps/compare_marker_to_automatic.cpp)
target_link_libraries(compare_marker_to_automatic PRIVATE manual_marker_core)
foreach(target generate_charuco_board calibrate_camera_markers estimate_marker_pose compare_marker_to_automatic)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()

add_executable(manual_marker_tests tests/manual_marker_tests.cpp)
target_link_libraries(manual_marker_tests PRIVATE manual_marker_core)
target_compile_options(manual_marker_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME manual_marker_tests COMMAND manual_marker_tests)
