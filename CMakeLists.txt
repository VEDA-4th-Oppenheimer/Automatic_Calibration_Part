cmake_minimum_required(VERSION 3.20)
project(auto_calib LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs imgproc)
find_package(Ceres REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)

add_library(synthetic_lidar src/synthetic_lidar.cpp src/calibration_core.cpp)
target_include_directories(synthetic_lidar PUBLIC include)
target_link_libraries(synthetic_lidar PUBLIC Eigen3::Eigen opencv_core opencv_imgcodecs opencv_imgproc Ceres::ceres yaml-cpp nlohmann_json::nlohmann_json)
target_compile_options(synthetic_lidar PRIVATE -Wall -Wextra -Wpedantic)

add_executable(generate_synthetic_scan apps/generate_synthetic_scan.cpp)
target_link_libraries(generate_synthetic_scan PRIVATE synthetic_lidar)
target_compile_options(generate_synthetic_scan PRIVATE -Wall -Wextra -Wpedantic)
add_executable(run_synthetic_calibration apps/run_synthetic_calibration.cpp)
target_link_libraries(run_synthetic_calibration PRIVATE synthetic_lidar)
add_executable(run_multi_synthetic_calibration apps/run_multi_synthetic_calibration.cpp)
target_link_libraries(run_multi_synthetic_calibration PRIVATE synthetic_lidar)
target_compile_options(run_multi_synthetic_calibration PRIVATE -Wall -Wextra -Wpedantic)
add_executable(render_calibration_visualization apps/render_calibration_visualization.cpp)
target_link_libraries(render_calibration_visualization PRIVATE synthetic_lidar)
target_compile_options(render_calibration_visualization PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(run_synthetic_calibration PRIVATE -Wall -Wextra -Wpedantic)

enable_testing()
add_executable(synthetic_lidar_tests tests/synthetic_lidar_tests.cpp)
target_link_libraries(synthetic_lidar_tests PRIVATE synthetic_lidar)
target_compile_options(synthetic_lidar_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME synthetic_lidar_tests COMMAND synthetic_lidar_tests)
add_executable(calibration_core_tests tests/calibration_core_tests.cpp)
target_link_libraries(calibration_core_tests PRIVATE synthetic_lidar)
target_compile_options(calibration_core_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME calibration_core_tests COMMAND calibration_core_tests)
